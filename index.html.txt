<script>
let scenario = 'Base';
let cfChart = null;

const SCENARIO_FACTORS = {
  Low:  { faToSL: 0.002, biasToSL: 0.001, trueLostSales: 0.30, faToInv: 0.005,  biasToInv: 0.002,  obsShare: 0.5, rushShare: 0.05, rushPrem: 1.5, expShare: 0.05, expPrem: 2.0 },
  Base: { faToSL: 0.003, biasToSL: 0.002, trueLostSales: 0.33, faToInv: 0.0075, biasToInv: 0.00305, obsShare: 0.6, rushShare: 0.10, rushPrem: 2.0, expShare: 0.075, expPrem: 2.5 },
  High: { faToSL: 0.004, biasToSL: 0.003, trueLostSales: 0.40, faToInv: 0.010,  biasToInv: 0.005,  obsShare: 0.7, rushShare: 0.15, rushPrem: 2.5, expShare: 0.10, expPrem: 3.0 }
};

const INV_DEFAULTS = {
  sw: [1200000, 350000, 350000, 350000],
  si: [2600000, 800000, 250000, 0],
  data: [900000, 250000, 100000, 50000],
  cm: [450000, 200000, 100000, 50000],
  internal: [650000, 350000, 150000, 100000],
  hypercare: [0, 200000, 250000, 250000],
};

function g(id) { return parseFloat(document.getElementById(id)?.value) || 0; }
function gs(id) { return document.getElementById(id)?.value || ''; }
function fmt(n) { const c = gs('b-currency') || 'EUR'; return c + ' ' + Math.round(n).toLocaleString(); }
function fmtM(n) { const c = gs('b-currency') || 'EUR'; return c + ' ' + (Math.round(n/100000)/10).toFixed(1) + 'M'; }
function pct(n) { return (Math.round(n*10)/10).toFixed(1) + '%'; }

function setScenario(s) {
  scenario = s;
  ['Low','Base','High'].forEach(x => {
    const b = document.getElementById('sb-' + x.toLowerCase());
    b.className = 'scen-btn';
  });
  document.getElementById('sb-' + s.toLowerCase()).className = 'scen-btn s-' + s.toLowerCase();
  recalc();
}

function calcModel() {
  const ns = g('b-netsales'), gm = g('b-gm')/100;
  const inv = g('b-inv'), dio = g('b-dio');
  const wo = g('b-writeoffs'), prod = g('b-prod'), trans = g('b-transport');
  const faBasePct = g('b-fa')/100, faimp = g('b-faimp'), bias = g('b-bias');
  const planner = g('b-planner'), contractor = g('b-contractor');
  const disc = g('b-disc')/100, r1 = g('b-r1')/100, r2 = g('b-r2')/100;
  const useGM = gs('b-usegm') === 'yes';
  const sf = SCENARIO_FACTORS[scenario];

  const relErrReduction = faimp / (1 - faBasePct);

  const revAtRisk = ns * 0.025;
  const slUpliftFA = faimp * sf.faToSL;
  const slUpliftBias = bias * sf.biasToSL;
  const gmFactor = useGM ? gm : 1;

  const D1 = ns * slUpliftFA * sf.trueLostSales * gmFactor;
  const D2 = ns * slUpliftBias * sf.trueLostSales * gmFactor;
  const D3_invFA = inv * faimp * sf.faToInv;
  const D4_invBias = inv * bias * sf.biasToInv;
  const D5_obs = wo * relErrReduction * sf.obsShare;
  const D6_prod = prod * relErrReduction * sf.rushShare * (sf.rushPrem - 1);
  const D7_log = trans * relErrReduction * sf.expShare * (sf.expPrem - 1);
  const D8_planner = planner;
  const D9_contractor = contractor;

  const annualRecurring = D1 + D2 + D5_obs + D6_prod + D7_log + D8_planner + D9_contractor;
  const oneTimeWC = D3_invFA + D4_invBias;

  const invTotals = calcInvTotals();
  const totalInv = invTotals.total3yr;
  const y0inv = invTotals.y0, y1inv = invTotals.y1, y2inv = invTotals.y2, y3inv = invTotals.y3;

  const cf = [
    -(y0inv),
    annualRecurring * r1 + oneTimeWC - y1inv,
    annualRecurring * r2 - y2inv,
    annualRecurring * 1.0 - y3inv,
  ];
  const dfs = [1, 1/(1+disc), 1/Math.pow(1+disc,2), 1/Math.pow(1+disc,3)];
  const npv = cf.reduce((s,v,i) => s + v*dfs[i], 0);
  const cum = cf.reduce((acc,v) => { acc.push((acc[acc.length-1]||0)+v); return acc; }, []);
  let payback = Infinity;
  for (let i=1; i<cum.length; i++) {
    if (cum[i]>=0) { payback = (i-1) + (-cum[i-1])/(cum[i]-cum[i-1]); break; }
  }
  const roi = y0inv > 0 ? npv / y0inv : 0;

  return { ns, gm, inv, dio, wo, prod, trans, planner, contractor, disc, r1, r2,
    D1, D2, D3_invFA, D4_invBias, D5_obs, D6_prod, D7_log, D8_planner, D9_contractor,
    annualRecurring, oneTimeWC, totalInv, y0inv, y1inv, y2inv, y3inv,
    cf, cum, npv, payback, roi, sf, dfs, relErrReduction, faimp, bias, useGM, faBasePct };
}

function calcInvTotals() {
  const cats = ['sw','si','data','cm','internal','hypercare'];
  let y0=0,y1=0,y2=0,y3=0;
  cats.forEach(c => {
    const el0 = document.getElementById('inv-'+c+'-0');
    const d = INV_DEFAULTS[c];
    const v = (i) => el0 ? (parseFloat(document.getElementById('inv-'+c+'-'+i)?.value)||0) : d[i];
    y0 += v(0); y1 += v(1); y2 += v(2); y3 += v(3);
  });
  const cont = (v) => Math.round(v*0.1);
  y0 += cont(y0); y1 += cont(y1); y2 += cont(y2); y3 += cont(y3);
  return { y0, y1, y2, y3, total3yr: y0+y1+y2+y3 };
}

}

async function ask(q) {
  document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.tab').forEach(t => { if(t.textContent.includes('AI')) t.classList.add('active'); else t.classList.remove('active'); });
  document.getElementById('tab-ai').classList.add('active');
  addMsg('user', q);
  const lid = addMsg('ai', '<span class="dot">Thinking…</span>');
  try {
    const r = await fetch('https://api.anthropic.com/v1/messages', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ model:'claude-sonnet-4-20250514', max_tokens:1000,
        system: buildContext(), messages: [...chatHistory, {role:'user',content:q}] })
    });
    const d = await r.json();
    const t = d.content?.map(b => b.text||'').join('') || 'No response received.';
    chatHistory.push({role:'user',content:q});
    chatHistory.push({role:'assistant',content:t});
    updMsg(lid, t);
  } catch(e) { updMsg(lid, 'Error reaching the AI. Please try again.'); }
}

async function send() {
  const inp = document.getElementById('chat-inp');
  const q = inp.value.trim(); if (!q) return;
  inp.value = '';
  await ask(q);
}

let mid = 0;
function addMsg(role, html) {
  const area = document.getElementById('chat-area');
  const id = 'msg-' + (++mid);
  const pfx = role === 'user'
    ? '<span style="font-size:10px;font-weight:500;color:var(--color-text-secondary);margin-right:6px">YOU</span>'
    : '<span style="font-size:10px;font-weight:500;color:#185FA5;margin-right:6px">AGENT</span>';
  area.innerHTML += `<div id="${id}" style="margin-bottom:8px;color:${role==='user'?'var(--color-text-secondary)':'var(--color-text-primary)'}">${pfx}${html}</div>`;
  area.scrollTop = area.scrollHeight;
  return id;
}
function updMsg(id, html) {
  const el = document.getElementById(id);
  if (el) {
    el.innerHTML = '<span style="font-size:10px;font-weight:500;color:#185FA5;margin-right:6px">AGENT</span>' + html.replace(/\n/g,'<br>');
    document.getElementById('chat-area').scrollTop = 99999;
  }
}
